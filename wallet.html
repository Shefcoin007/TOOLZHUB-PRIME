// Korapay Payment Link const KORAPAY_LINK = 'https://test-checkout.korapay.com/pay/QKaF9T6aabNMaKF'; document.addEventListener('DOMContentLoaded', async () => { const { data: { session } } = await supabaseClient.auth.getSession(); if (!session) { window.location.href = 'index.html'; return; } currentUser = session.user; await loadWallet(); }); async function loadWallet() { try { // Load profile const { data: profile } = await supabaseClient .from('profiles') .select('balance, points') .eq('id', currentUser.id) .single(); if (profile) { document.getElementById('userBalance').textContent = (profile.balance || 0).toFixed(2); document.getElementById('userPoints').textContent = profile.points || 0; } // Load transactions await loadTransactions(); } catch (err) { console.error('Wallet load error:', err); } } async function loadTransactions() { try { const { data: txs } = await supabaseClient .from('transactions') .select('*') .eq('user_id', currentUser.id) .order('created_at', { ascending: false }) .limit(20); const container = document.getElementById('transactionsList'); if (!container) return; if (!txs || txs.length === 0) { container.innerHTML = '

No transactions yet

'; return; } container.innerHTML = txs.map(tx => `
${tx.type.toUpperCase()} ${new Date(tx.created_at).toLocaleString()} ${tx.method ? `${tx.method}` : ''}
${tx.type === 'purchase' || tx.type === 'withdrawal' ? '-' : '+'}₦${tx.amount.toLocaleString()}
${tx.status}
`).join(''); } catch (err) { console.error('Error loading transactions:', err); } } // Handle fund request form async function handleFundSubmit(e) { e.preventDefault(); const amount = parseFloat(document.getElementById('fundAmount')?.value); const method = document.getElementById('fundMethod')?.value || 'bank'; const reference = document.getElementById('fundReference')?.value; if (!amount || amount < 1000) { showToast('Minimum amount is ₦1,000', 'error'); return; } try { // Record transaction request await supabaseClient.from('transactions').insert({ user_id: currentUser.id, type: 'deposit', amount: amount, method: method, reference: reference || `MANUAL-${Date.now()}`, status: 'pending' }); showToast('Fund request submitted! Admin will verify shortly.', 'success'); document.getElementById('fundForm')?.reset(); loadTransactions(); } catch (err) { showToast('Failed: ' + err.message, 'error'); } } // Open Korapay payment function openKorapayPayment() { const amount = document.getElementById('fundAmount')?.value; if (!amount || amount < 1000) { showToast('Enter amount (minimum ₦1,000)', 'error'); return; } // Store pending transaction localStorage.setItem('pendingPayment', JSON.stringify({ amount: amount, timestamp: Date.now(), userId: currentUser.id })); // Open Korapay in new window window.open(KORAPAY_LINK, '_blank'); showToast('Complete payment on Korapay. Your balance will update automatically.', 'success'); // Check for payment completion after 10 seconds setTimeout(checkPaymentStatus, 10000); } async function checkPaymentStatus() { // Reload wallet to check for new balance await loadWallet(); showToast('Wallet refreshed. Check if your balance updated.', 'success'); } async function redeemPoints() { const points = parseInt(document.getElementById('userPoints')?.textContent || '0'); if (points < 1) { showToast('You need at least 1 point to redeem', 'error'); return; } if (!confirm(`Redeem ${points} points for ₦${points * 100}?`)) return; try { await supabaseClient.from('points_redemptions').insert({ user_id: currentUser.id, points_used: points, value_received: points * 100, status: 'completed' }); const { data: profile } = await supabaseClient .from('profiles') .select('balance') .eq('id', currentUser.id) .single(); await supabaseClient .from('profiles') .update({ balance: (profile?.balance || 0) + (points * 100), points: 0 }) .eq('id', currentUser.id); showToast(`Redeemed! ₦${points * 100} added to balance.`, 'success'); loadWallet(); } catch (err) { showToast('Failed: ' + err.message, 'error'); } } async function logout() { await supabaseClient.auth.signOut(); window.location.href = 'index.html'; } // Expose functions globally window.handleFundSubmit = handleFundSubmit; window.openKorapayPayment = openKorapayPayment; window.redeemPoints = redeemPoints; window.logout = logout;
